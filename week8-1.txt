#include <iostream>
#include <string>

using namespace std;

int main() {
    string inputStr;
    string accumulatedStr = "";

    while (true) {
        cout << "문자열을 입력하시오 : ";
        cin >> inputStr;

        // exit 입력 시 종료
        if (inputStr == "exit") {
            cout << "exit가 입력되어 종료합니다." << "\n";
            break;
        }

        // 문자열 누적 및 출력
        accumulatedStr += inputStr;
        cout << "누적 문자열 -> " << accumulatedStr << "\n";

        // 누적 문자 길이가 200자를 초과할 경우 종료
        if (accumulatedStr.length() > 200) {
            cout << "누적 200자가 넘어서 종료합니다." << "\n";
            break;
        }
    }

    return 0;
}
