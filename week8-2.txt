#include <iostream>
#include <string>

using namespace std;

int main() {
    // 템플릿 코드에서 파일을 읽어 무작위로 생성한 정답이 들어갈 변수입니다.
    // 템플릿의 정답 변수명에 맞춰 아래 문자열을 교체해 주세요.
    string answer = "crane"; 
    
    bool isWin = false;

    // 최대 6번의 추측 기회 제공
    for (int i = 1; i <= 6; ++i) {
        string guess;
        cout << "guess " << i << ": ";
        cin >> guess;

        string result = "";

        // 입력한 단어의 각 위치와 문자 비교
        for (int j = 0; j < 5; ++j) {
            if (guess[j] == answer[j]) {
                result += "G"; // 위치와 문자가 모두 일치
            } else if (answer.find(guess[j]) != string::npos) {
                result += "Y"; // 문자는 존재하지만 위치가 다름
            } else {
                result += "-"; // 정답에 없는 문자
            }
        }

        cout << result << "\n";

        // 정답을 맞힌 경우
        if (result == "GGGGG") {
            cout << "you win!" << "\n";
            isWin = true;
            break;
        }
    }

    // 6번 안에 맞히지 못한 경우
    if (!isWin) {
        cout << "you lose! answer is: " << answer << "\n";
    }

    return 0;
}
